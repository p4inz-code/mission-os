# mission-ui — Mission OS UI Component Library
#
# This library provides reusable QML components, design tokens,
# theme support, and accessibility wrappers for all Mission OS
# applications.
#
# Stack: Qt 6, KDE Kirigami, KDE Frameworks
#
# Module: org.mission.ui (QML import)

    VERSION 0.1.0
    DESCRIPTION "Mission OS UI Component Library"
    LANGUAGES CXX

# ── QML Resources ───────────────────────────────────────────────────
# mission-ui is a library module: NO_RESOURCE_TARGET_PATH is not valid
# for a non-executable backing target (Qt6QmlMacros FATAL_ERROR on
# Qt >= 6.10), so it is omitted. The module is consumed from the build
# tree via the QML import path (OUTPUT_DIRECTORY below) and at runtime
# via the installed plugin, not from an embedded resource path.
#
# Adopt the Qt QTP0001 NEW default (resource prefix '/qt/qml' without a
# version segment) to keep the configure log clean; qt_policy is absent
# on Qt < 6.6.
if(COMMAND qt_policy)
    qt_policy(SET QTP0001 NEW)
endif()
qt6_add_qml_module(mission-ui
    URI org.mission.ui
    # Public QML API version: must match the version used by imports
    # (`import org.mission.ui 1.0` in SmokeTest.qml / tests) — the
    # generated qmldir registers types at this version.
    VERSION 1.0
    QML_FILES ${MISSION_UI_QML_FILES}
    # Deterministic build-tree layout: OUTPUT_DIRECTORY must BE the module
    # directory (it ends in the module target path org/mission/ui), so the
    # compiled module (generated qmldir, plugin) lands at
    # <binary>/qml/org/mission/ui/ and the QML import path for tests and
    # tooling is <binary>/qml.
    OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/qml/org/mission/ui
)

# ── Build-tree qmldir post-processing ──────────────────────────────
# The qmldir generated by qt6_add_qml_module lists the pragma-Singleton
# token files without the `singleton` marker when the toolchain lacks the
# QML compiler (qmlcachegen/qmlsc — e.g. Ubuntu), and its
# `prefer :/qt/qml/...` module-directory-map directive breaks runtime
# singleton initialization in Qt 6.10. fix-module-qmldir.cmake rewrites
# the generated build-tree qmldir (idempotently) to add the markers and
# drop the prefer line so the module loads correctly on every toolchain.
# The singleton list is hardcoded inside the script (fixed module API).
add_custom_command(TARGET mission-ui POST_BUILD
    COMMAND ${CMAKE_COMMAND}
        -DQMLDIR=${CMAKE_CURRENT_BINARY_DIR}/qml/org/mission/ui/qmldir
        -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/fix-module-qmldir.cmake
    COMMENT "mission-ui: rewriting generated qmldir (singleton markers, no prefer)"
    VERBATIM
)

    # ── Static validation: qmlsc --verify ─────────────────────────
    # Runs only where the installed Qt toolchain actually ships qmlsc
    # (Qt6::qmlsc). Ubuntu-based toolchains do not ship qmlsc, so there
    # the test reports SKIPPED explicitly instead of pretending to pass.
    if(Qt6QmlTools_FOUND AND COMMAND qt6_find_qmlsc)
        qt6_find_qmlsc()
    endif()
    if(TARGET Qt6::qmlsc)
        add_test(NAME mission-ui-qmlsc
            COMMAND Qt6::qmlsc --verify
                    --import-path "${MISSION_UI_TEST_IMPORT_PATH}"
                    "${CMAKE_CURRENT_SOURCE_DIR}/qml/org/mission/ui/SmokeTest.qml"
        )
        set_tests_properties(mission-ui-qmlsc PROPERTIES
            LABELS "mission-ui;qml;static"
            ENVIRONMENT "QT_QML_IMPORT_PATH=${MISSION_UI_TEST_IMPORT_PATH}"
        )
    else()
        add_test(NAME mission-ui-qmlsc
            COMMAND ${CMAKE_COMMAND} -E echo
                "SKIPPED: qmlsc (Qt6 QML compiler) is not available in this Qt toolchain — install a Qt build that ships qmlsc (Qt6::qmlsc) to enable static QML verification"
        )
